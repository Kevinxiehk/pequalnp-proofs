theorem blk_row_scale {n : ℕ} {m : Type} [Fintype m] [DecidableEq m] (ρ κ : m → Fin n)
    (d : Fin n → ℂ) (w : Fin n × Fin n → ℂ) :
    blk ρ κ (fun p => d p.1 * w p) = (∏ i, d (ρ i)) * blk ρ κ w := by
  unfold blk
  simp only [Matrix.permanent, Matrix.of_apply]
  rw [Finset.mul_sum]
  refine Finset.sum_congr rfl fun σ _ => ?_
  rw [Finset.prod_mul_distrib]
  congr 1
  exact Equiv.prod_comp σ (fun i => d (ρ i))

theorem t1_identity {m : ℕ} {L : Submodule ℂ (Fin (m + 2) × Fin (m + 2) → ℂ)}
    (hL : Lev (m + 2) L) (e : Fin (m + 2))
    (hrank : m + 1 ≤ Module.finrank ℂ (L.map (rowMap e)))
    (S : Finset (Fin (m + 2))) (heS : e ∉ S)
    (x ξ : Fin (m + 2) × Fin (m + 2) → ℂ) (hx : x ∈ L) (hξ : ξ ∈ L)
    (hξe : ∀ j, ξ (e, j) = 0) (hxS : ∀ i, i ∉ S → ∀ j, x (i, j) = 0) (j : Fin (m + 2)) :
    blk e.succAbove j.succAbove (fun p => if p.1 ∈ S then x p else ξ p) = 0 := by
  obtain ⟨P, hP⟩ := polyLine_blk e.succAbove j.succAbove
    (fun p : Fin (m + 2) × Fin (m + 2) => if p.1 ∈ S then x p else ξ p)
    (fun p : Fin (m + 2) × Fin (m + 2) => if p.1 ∈ S then ξ p else 0)
  have hz : ∀ s : ℂ, s ≠ 0 → P.eval s = 0 := by
    intro s hs
    rw [← hP s]
    have e1 : ((fun p : Fin (m + 2) × Fin (m + 2) => if p.1 ∈ S then x p else ξ p) +
        s • fun p : Fin (m + 2) × Fin (m + 2) => if p.1 ∈ S then ξ p else 0) =
        fun p => (if p.1 ∈ S then s else 1) * (ξ + s⁻¹ • x) p := by
      funext p
      rcases p with ⟨i, j'⟩
      by_cases hp : i ∈ S
      · simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul, if_pos hp]
        rw [mul_add, ← mul_assoc, mul_inv_cancel₀ hs, one_mul]
        ring
      · simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul, if_neg hp]
        rw [hxS i hp j']
        ring
    rw [e1]
    have h3 := ca2_full hL e hrank (ξ + s⁻¹ • x) (L.add_mem hξ (L.smul_mem _ hx))
      (fun j' => by
        simp only [Pi.add_apply, Pi.smul_apply, smul_eq_mul]
        rw [hξe j', hxS e heS j']
        ring) j
    have h4 := blk_row_scale e.succAbove j.succAbove (fun i => if i ∈ S then s else 1)
      (ξ + s⁻¹ • x)
    exact h4.trans (by rw [h3, mul_zero])
  have hP0 := poly_zero_of_ne P hz
  have h0 := hP 0
  rw [zero_smul, add_zero, hP0, Polynomial.eval_zero] at h0
  exact h0
